function getRange(price){
  if(price === 50)  return [51,100];
  if(price === 100) return [41,80];
  if(price === 200) return [21,60];
  if(price === 500) return [1,40];
  return [1,100];
}
function pickFromBoard(boardKey, price){
  const range = getRange(price);
  const all = DATA[boardKey] || [];
  let pool = all.filter(x => x[0] >= range[0] && x[0] <= range[1]);
  if(pool.length === 0) pool = all;
  return pool[Math.floor(Math.random() * pool.length)];
}
function pickByRankRange(boardKey, minRank, maxRank){

  let pool = all.filter(x => x[0] >= minRank && x[0] <= maxRank);
  if(pool.length === 0) pool = all;
  return pool[Math.floor(Math.random() * pool.length)];
}

/* ============================================================

   ============================================================ */
// 基础血量：排名越靠前越高。第1名 = 1000，第100名 = 100
function calcBaseHp(rank){
  return Math.round(100 + (100 - rank) * 9.09);
}
// 基础攻击：武力值 × 3
function calcBaseAtk(value){
  return Math.round(value * 3);
}
// 基础防御：武力值 × 1.5
function calcBaseDef(value){
  return Math.round(value * 1.5);
}
// 基础暴击率：武力值 × 0.5%（上限100%）
function calcBaseCritRate(value){
  return Math.min(100, Math.round(value * 0.5 * 10) / 10);
}
// 基础暴击伤害：100% + 武力值 × 2%（上限500%）
function calcBaseCritDmg(value){
  return Math.min(500, Math.round((100 + value * 2) * 10) / 10);
}

function buildOwnedItem(boardKey, row){
  const b = BOARDS.find(x => x.key === boardKey);
  return {
    boardKey, rank:row[0], dynasty:row[1], emperor:row[2], name:row[3], value:row[4],
    label:b.label, type:b.type,
    critRate: calcBaseCritRate(row[4]),
    critDmg: calcBaseCritDmg(row[4])
  };
}
function getItemStats(item){
  return {
    hp: calcBaseHp(item.rank),
    atk: calcBaseAtk(item.value),
    def: calcBaseDef(item.value),
    critRate: item.critRate,
    critDmg: item.critDmg
  };
}

function addToken(name, dynasty, emperor){
  let t = state.tokens.find(x => x.name === name);
  if(t){ t.count++; }
  else { state.tokens.push({ name, dynasty, emperor, count:1 }); }
}
function removeToken(name){
  const idx = state.tokens.findIndex(x => x.name === name);
  if(idx >= 0){
    state.tokens[idx].count--;
    if(state.tokens[idx].count <= 0) state.tokens.splice(idx, 1);
    return true;
  }
  return false;
}

function confirmGiveToken(){
  const i = window._selectedTokenIdx;
  if(i < 0){ alert("请选择要赠送的信物！"); return; }
  const token = state.tokens[i];
  if(!token){ alert("信物不存在！"); return; }
  const type = document.getElementById("tokenBonusType").value;
  const item = currentDetailItem;

  // 上限检查
  if(type === "critRate" && item.critRate >= 100){
    // 暴击率已满 → 转积分
    state.points += 10;
    removeToken(token.name);
    addLog(`【${item.name}】暴击率已满，信物转化为10积分`);
    renderTop(); save();
    closeModal("tokenModal");
    showResult("🎁 信物转化", `<div class="result-card"><div class="n">暴击率已满</div><div class="v">转化为 10 积分</div></div>`, `<span class="win">转化完成</span>`);
    return;
  }
  if(type === "critDmg" && item.critDmg >= 500){
    state.points += 10;
    removeToken(token.name);
    addLog(`【${item.name}】暴伤已满，信物转化为10积分`);
    renderTop(); save();
    closeModal("tokenModal");
    showResult("🎁 信物转化", `<div class="result-card"><div class="n">暴击伤害已满</div><div class="v">转化为 10 积分</div></div>`, `<span class="win">转化完成</span>`);
    return;
  }

  // 加成
  if(type === "critRate"){
    item.critRate = Math.min(100, item.critRate + TOKEN_CRIT_RATE);
  } else {
    item.critDmg = Math.min(500, item.critDmg + TOKEN_CRIT_DMG);
  }
  removeToken(token.name);
  addLog(`【${item.name}】赠送${token.name}信物 → ${type === "critRate" ? "暴击率+5%" : "暴伤+20%"}`);
  save();
  closeModal("tokenModal");
  const s = getItemStats(item);
  showResult("🎁 赠送成功",
    `<div class="result-card"><div class="r">${item.name}</div>
     <div class="v">暴击率：${s.critRate}% / 100%</div>
     <div class="v">暴击伤害：${s.critDmg}% / 500%</div></div>`,
    `<span class="win">赠送成功！</span>`);
  // 重新渲染详情
  const idx = state.owned.indexOf(item);
  if(idx >= 0) openDetail(idx);
}

function openPick(mode, battleMode, diffKey){
  const boardKey = battleMode ? BATTLE_CONFIG[battleMode].boardKey : null;
  const wuBoards = ["force","command","overall"];
  const mouBoards = ["intel","logistics","advOverall"];
  let pool, title, hint;
  if(mode === "daily"){
    pool = state.owned.filter(o => o.boardKey === boardKey);
    title = "选择出战 · " + BATTLE_CONFIG[battleMode].title;
    hint = "从你的收藏中选择一名" + BATTLE_CONFIG[battleMode].type + "出战";
  } else {
    pool = state.owned.filter(o => wuBoards.includes(o.boardKey) || mouBoards.includes(o.boardKey));
    title = "选择出战 · 国运之战";
    hint = "需要一名武将 + 一名谋士（共选两个）";
  }
  if(pool.length === 0){
    alert("没有可用名臣！请先去抽取。");
    return;
  }
  pickContext = { mode, battleMode, diffKey, wuPick:null, mouPick:null, pool };
  document.getElementById("pickTitle").textContent = title;
  document.getElementById("pickHint").textContent = hint;
  renderPickList();
  openModal("pickModal");
}
function renderPickList(){
  const el = document.getElementById("pickList");
  const pc = pickContext;
  const wuBoards = ["force","command","overall"];
  let html = "";
  pc.pool.forEach((o, idx) => {
    const isWu = wuBoards.includes(o.boardKey);
    let cls = "pick-item";
    if(pc.mode === "main"){
      if(isWu && pc.wuPick === o) cls += " selected";
      if(!isWu && pc.mouPick === o) cls += " selected";
    } else {
      if(pc.wuPick === o) cls += " selected";
    }
    const typeTag = isWu ? "武将" : "谋士";
    html += `<div class="${cls}" onclick="togglePick(${idx})">
      <div>
        <div class="pi-name">${o.dynasty}_${o.emperor}_${o.name}</div>
        <div style="color:#888;font-size:11px">${o.label} · 第${o.rank}名 · ${typeTag}</div>
      </div>
      <div class="pi-val">${o.value}</div>
    </div>`;
  });
  el.innerHTML = html;
}
function togglePick(idx){
  const pc = pickContext;
  const o = pc.pool[idx];
  const wuBoards = ["force","command","overall"];
  const isWu = wuBoards.includes(o.boardKey);
  if(pc.mode === "daily"){
    pc.wuPick = o;
  } else {
    if(isWu){ pc.wuPick = o; }
    else { pc.mouPick = o; }
  }
  renderPickList();
}
function confirmPick(){

  if(pc.mode === "daily"){

    closeModal("pickModal");
    startMainQuestWithPick(pc.wuPick, pc.mouPick);
  }
}

        <div style="font-size:11px;color:#888">HP <span id="oppHpText">${oStats.hp}</span>/${oStats.hp}</div>
      </div>
    </div>
  `;
  const log = document.getElementById("roundLog");
  log.innerHTML = "";
  log.innerHTML += `<div class="rinfo">🎲 投掷骰子：${dice} 点（${dice % 2 === 1 ? "单数" : "双数"}）→ ${first === "player" ? "我方先手" : "对方先手"}</div>`;
  document.getElementById("duelResultText").innerHTML = "";
  document.getElementById("duelNextBtn").style.display = "inline-block";
  document.getElementById("duelNextBtn").textContent = "下一回合";
  document.getElementById("duelNextBtn").onclick = duelNext;
  document.getElementById("duelCloseBtn").style.display = "none";
  log.scrollTop = log.scrollHeight;
  openModal("duelModal");
}

function calcDamage(atk, def, critRate, critDmg){
  let dmg = Math.max(5, Math.round(atk - def * 0.5));
  let isCrit = Math.random() * 100 < critRate;
  if(isCrit){
    dmg = Math.round(dmg * critDmg / 100);
  }
  return { dmg, isCrit };
}

function duelNext(){
  if(!duelContext || duelContext.finished) return;
  const dc = duelContext;
  dc.round++;
  const log = document.getElementById("roundLog");

  if(dc.turn === "player"){
    const r = calcDamage(dc.pStats.atk, dc.oStats.def, dc.pStats.critRate, dc.pStats.critDmg);
    dc.oppHp = Math.max(0, dc.oppHp - r.dmg);
    const cls = r.isCrit ? "rcrit" : "rwin";
    const tag = r.isCrit ? "【暴击】" : "";
    log.innerHTML += `<div class="${cls}">第${dc.round}回合 · 我方${tag}出手：造成 ${r.dmg} 伤害（对方剩余HP ${dc.oppHp}）</div>`;
    dc.turn = "opp";
  } else {
    const r = calcDamage(dc.oStats.atk, dc.pStats.def, dc.oStats.critRate, dc.oStats.critDmg);
    dc.playerHp = Math.max(0, dc.playerHp - r.dmg);
    const cls = r.isCrit ? "rcrit" : "rlose";
    const tag = r.isCrit ? "【暴击】" : "";
    log.innerHTML += `<div class="${cls}">第${dc.round}回合 · 对方${tag}出手：造成 ${r.dmg} 伤害（我方剩余HP ${dc.playerHp}）</div>`;
    dc.turn = "player";
  }
  updateHpBars();
  log.scrollTop = log.scrollHeight;

  if(dc.playerHp <= 0 || dc.oppHp <= 0){
    dc.finished = true;
    finishDuel();
  }
}

function updateHpBars(){
  const dc = duelContext;
  const pPct = (dc.playerHp / dc.playerHpMax) * 100;
  const oPct = (dc.oppHp / dc.oppHpMax) * 100;
  document.getElementById("playerHpBar").style.width = pPct + "%";
  document.getElementById("oppHpBar").style.width = oPct + "%";
  document.getElementById("playerHpText").textContent = dc.playerHp;
  document.getElementById("oppHpText").textContent = dc.oppHp;
}

function finishDuel(){
  const dc = duelContext;
  const log = document.getElementById("roundLog");
  const win = dc.oppHp <= 0 && dc.playerHp > 0;
  let gainText = "";

  if(dc.mode === "daily"){
    if(win){
      const back = Math.floor(dc.fee * dc.mult);
      state.points += back;
      gainText = `<span class="win">胜利！获得 ${back} 积分（净赚 ${back - dc.fee}）</span>`;
    } else {
      gainText = `<span class="lose">败北！损失 ${dc.fee} 积分</span>`;
    }
  } else {
    // 国运之战：不花积分，改排行榜积分
    const delta = win ? 30 : -15;
    state.leaderboard["华夏"] += delta;
    state.leaderboard["？？？"] += win ? -15 : 20;
    if(state.leaderboard["华夏"] < 0) state.leaderboard["华夏"] = 0;
    if(state.leaderboard["？？？"] < 0) state.leaderboard["？？？"] = 0;
    gainText = win
      ? `<span class="win">国运昌隆！华夏 +30，？？？ -15</span>`
      : `<span class="lose">国运受挫！华夏 -15，？？？ +20</span>`;
  }

  log.innerHTML += `<div class="rinfo">🏆 决斗结束：${win ? "我方获胜" : "我方战败"}</div>`;
  document.getElementById("duelResultText").innerHTML = gainText;
  document.getElementById("duelNextBtn").style.display = "none";
  document.getElementById("duelCloseBtn").style.display = "inline-block";
  renderTop(); save();
  addLog(`【${dc.title}】${dc.diffName} → 我方${dc.player.name} vs 对手${dc.opp.name} → ${win?"胜":"负"}`);
}

}

function adminAddCharacter(){
  const boardKey = document.getElementById("adminCharBoard").value;
  const rank = parseInt(document.getElementById("adminCharRank").value);
  if(!rank || rank < 1 || rank > 100){ alert("请输入1-100的排名"); return; }
  const all = DATA[boardKey] || [];
  const row = all.find(x => x[0] === rank);
  if(!row){ alert("当前榜单暂无第" + rank + "名数据"); return; }
  if(isOwned(boardKey, row[0])){
    addToken(row[3], row[1], row[2]);
    addLog(`管理员添加重复：${row[3]} → 信物×1`);
    alert("已拥有该名臣，获得信物×1！");
  } else {
    state.owned.push(buildOwnedItem(boardKey, row));
    addLog(`管理员添加：${row[3]}（第${row[0]}名）`);
    alert("已添加到收藏！");
  }
  save(); renderOwned();
}
